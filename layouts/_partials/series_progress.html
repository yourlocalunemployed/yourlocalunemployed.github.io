Same query as the box, so the two cannot disagree. */ -}} {{- with .Params.series }} {{- $name := index . 0 }} {{- $posts := sort (where site.RegularPages "Params.series" "intersect" (slice $name)) "Date" "asc" }} {{- if gt (len $posts) 1 }} {{- $current := $.Permalink }} {{- $idx := 0 }} {{- range $i, $p := $posts }}{{ if eq $p.Permalink $current }}{{ $idx = (add $i 1) }}{{ end }}{{ end }} {{- $hub := site.GetPage (printf "series/%s" ($name | urlize)) }}
{{- with $hub }}{{ $name }}{{ else }}{{ $name }}{{ end }}
    {{- range $i, $p := $posts }} {{- $n := add $i 1 }}
  1. Part {{ $n }}: {{ $p.Params.seriesTitle | default $p.Title }}
    2. {{- end }}
part {{ $idx }} of {{ len $posts }}
{{- end }} {{- end }}
